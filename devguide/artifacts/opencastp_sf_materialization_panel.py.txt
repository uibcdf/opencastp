"""Audit three declared input conventions over all 1MRG SA atom contributions."""

import csv
import ctypes as C
import hashlib
import json
import pickle
import sys
import time
from importlib import metadata
from pathlib import Path
from zipfile import ZipFile

import numpy as np

from opencastp._core import volbl
from opencastp._core.geometry import (
    ALF_VERTEX, ALF_EDGE, ALF_TRIANGLE, ALF_TETRA,
    _weighted_hidden0, _weighted_hidden1,
)

geometry = pickle.loads(Path('/tmp/opencastp_1mrg_geometry_314.pickle').read_bytes())
archive_path = Path('/home/diego/repos@uibcdf/topomt/topomt/data/CASTpFold_server/1mrg.zip')
with ZipFile(archive_path) as arc:
    raw_pdb = arc.read(next(n for n in arc.namelist() if n.endswith('.pdb')))
    raw_csv = arc.read(next(n for n in arc.namelist() if n.endswith('.contrib.csv')))
points_to_ids = {}
for line in raw_pdb.decode().splitlines():
    if line.startswith('ATOM'):
        point = tuple(float(line[a:b]) for a, b in [(30, 38), (38, 46), (46, 54)])
        assert point not in points_to_ids
        points_to_ids[point] = int(line[6:11])
ids = [points_to_ids[tuple(round(float(v), 3) for v in p)] for p in geometry.atom_coordinates]
targets = {int(row['Atom_ID']): row for row in csv.DictReader(line for line in raw_csv.decode().splitlines() if not line.startswith('#'))}
assert set(ids) == set(targets)
expected = np.array([[float(targets[i]['SA_Volume']), float(targets[i]['SA_Area'])] for i in ids])
edges = volbl._edge_atoms_by_master_id(geometry)
faces = volbl._face_atoms_by_master_id(geometry)
terms = []
for rank in range(1, geometry.base_rank+1):
    bounds = geometry.master_rank_offsets.get(rank)
    if bounds is None:
        continue
    for entry in geometry.master_entries[bounds[0]:bounds[1]]:
        kind = int(entry.f_type)
        if kind != ALF_TETRA and not entry.is_first:
            continue
        if kind == ALF_VERTEX:
            atoms = [int(entry.index)]
        elif kind == ALF_EDGE:
            atoms = list(edges[int(entry.index)])
        elif kind == ALF_TRIANGLE:
            atoms = list(faces[int(entry.index)])
        else:
            atoms = list(map(int, geometry.mesh.simplex_atom_indices[int(entry.index)]))
        for atom in atoms:
            ordered = [atom, *[a for a in atoms if a != atom]]
            terms.append((len(atoms)-1, atom, [*ordered, *[0]*(4-len(ordered))]))
H0 = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int)
H1 = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int, C.c_int)
CCW = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int, C.c_int, C.c_int)
P = C.POINTER(C.c_double)
lib = C.CDLL('/tmp/opencastp_historical_sf.so')
lib.diagnostic_initialize.argtypes = [C.c_int, P, P, H0, H1, CCW]
lib.diagnostic_sf_term.argtypes = [C.c_int, C.c_int, C.c_int, C.c_int, C.c_int, P]
original, radii = geometry.atom_coordinates, geometry.atom_radii
weights = radii**2
h0 = H0(lambda i, j: int(_weighted_hidden0(original[i], float(weights[i]), original[j], float(weights[j])) != 0))
h1 = H1(lambda i, j, k: int(_weighted_hidden1(original[[i, j]], weights[[i, j]], original[k], float(weights[k])) != 0))
ccw = CCW(lambda i, j, k, l: int(np.linalg.det(np.array([original[j]-original[i], original[k]-original[i], original[l]-original[i]])) < 0))
scale = np.float32(1e-5)
fixed_points = np.rint(original*100000).astype(np.int64)
fixed_radii = np.rint(radii*100000).astype(np.int64)
legacy_points = (scale*fixed_points.astype(np.float32)).astype(np.float64)
legacy_radii = (scale*np.sqrt((fixed_radii**2).astype(np.float32))).astype(np.float64)
variants = {
    'baseline': (original, radii),
    'coordinates_direct_float32': (original.astype(np.float32).astype(float), radii),
    'both_alf_get_coords_float32': (legacy_points, legacy_radii),
}
report = {
    'case': '1mrg', 'atoms': len(ids), 'required_scalars': int(expected.size),
    'scope': 'SA space-filling per-atom CHECKING; unchanged historical C primitives, fixed modern topology/predicates',
    'units': {'SA_Volume': 'angstrom**3', 'SA_Area': 'angstrom**2'},
    'source_commit': '4ffb6cbf3273a0cf22c42f700137ee96e2eead8a',
    'python': sys.version.split()[0],
    'versions': {name: metadata.version(name) for name in ('numpy', 'scipy', 'opencastp')},
    'complete_server_equivalence': False, 'completed': False,
    'primitive_count': len(terms), 'variants': [],
    'prepared_coordinate_sha256': hashlib.sha256(original.tobytes()).hexdigest(),
    'prepared_expanded_radius_sha256': hashlib.sha256(radii.tobytes()).hexdigest(),
    'pdb_sha256': hashlib.sha256(raw_pdb).hexdigest(),
    'csv_sha256': hashlib.sha256(raw_csv).hexdigest(),
}
for variant, (points, sizes) in variants.items():
    start = time.monotonic()
    points = np.ascontiguousarray(points, dtype=np.float64)
    sizes = np.ascontiguousarray(sizes, dtype=np.float64)
    lib.diagnostic_initialize(len(sizes), points.ctypes.data_as(P), sizes.ctypes.data_as(P), h0, h1, ccw)
    values = np.zeros_like(expected)
    out = np.zeros(2)
    for kind, atom, indices in terms:
        lib.diagnostic_sf_term(kind, *indices, out.ctypes.data_as(P))
        values[atom] += out
    tolerances = np.array([0.00050001, 0.00500001])
    strict = np.abs(values-expected) <= tolerances
    exported = np.array([[np.round(float(f'{v:.4f}'), digits) for v, digits in zip(row, (3, 2))] for row in values])
    matches = exported == expected
    failures = []
    for atom, column in zip(*np.nonzero(~matches)):
        failures.append({'atom_id': ids[atom], 'field': ('SA_Volume', 'SA_Area')[column],
                         'actual': float(values[atom, column]), 'expected': float(expected[atom, column]),
                         'diagnostic_export': float(exported[atom, column]), 'strict_passed': bool(strict[atom, column])})
    result = {
        'variant': variant, 'strict_matched': int(strict.sum()),
        'export_matched': int(matches.sum()), 'export_failures': failures,
        'introduced_export_failures': int((strict & ~matches).sum()),
        'explained_original_failures': int((~strict & matches).sum()),
        'corrections': lib.diagnostic_corrections(), 'warnings': lib.diagnostic_warnings(),
        'elapsed_seconds': time.monotonic()-start,
    }
    report['variants'].append(result)
    Path('/tmp/opencastp_sf_materialization_panel.json').write_text(json.dumps(report, indent=2)+'\n')
    print({k: v for k, v in result.items() if k != 'export_failures'}, flush=True)
report['completed'] = True
Path('/tmp/opencastp_sf_materialization_panel.json').write_text(json.dumps(report, indent=2)+'\n')

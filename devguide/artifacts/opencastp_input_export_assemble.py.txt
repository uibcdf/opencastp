"""Assemble bounded input/export controls without redistributing external inputs."""

import hashlib
import json
import sys
from datetime import datetime, timezone
from importlib import metadata
from pathlib import Path
from zipfile import ZipFile

ROOT = Path('/home/diego/repos@uibcdf/opencastp')
LEGACY = Path('/home/diego/repos@uibcdf/Alphashape/castp')
ARCHIVES = Path('/home/diego/repos@uibcdf/topomt/topomt/data/CASTpFold_server')


def read(name):
    return json.loads(Path('/tmp', name).read_text())


collectors = [
    'opencastp_input_materialization_probe.py',
    'opencastp_float_metric_probe.py',
    'opencastp_atom_contribution_probe.py',
    'opencastp_atom588_metric_probe.py',
    'opencastp_historical_sf_bridge.c',
    'opencastp_atom588_materialization.py',
    'opencastp_sf_materialization_panel.py',
    'opencastp_fetch_raw_job_outputs.py',
    'opencastp_input_export_assemble.py',
]
controls = []
for name, real in [('opencastp_input_materialization_probe.json', 'double'), ('opencastp_float_metric_probe.json', 'float')]:
    source = read(name)
    cases = []
    for case in source['cases']:
        cases.append({
            'case': case['case'], 'field': case['field'],
            'unit': 'angstrom**2' if case['field'] == 'area_sa' else 'angstrom**3',
            'expected': case['expected'], 'variants': [
                {'variant': row['variant'], 'actual': row[case['field']], 'passed': row['passes'],
                 'corrections': row['corrections'], 'warnings': row['warnings'],
                 'max_coordinate_change_angstrom': row['max_coordinate_change_angstrom'],
                 'max_radius_change_angstrom': row['max_radius_change_angstrom']}
                for row in case['variants']
            ],
        })
    controls.append({'vol_real': real, 'metric_unchanged': real == 'double',
                     'fixed_modern_domains_and_predicates': True,
                     'assembly': 'double-valued wrapper terms accumulated with math.fsum', 'cases': cases})
atoms = read('opencastp_atom_contribution_probe.json')
atoms['units'] = {'SA_Area': 'angstrom**2', 'SA_Volume': 'angstrom**3'}
outputs = read('opencastp_raw_output_retrieval.json')['cases']
for row in outputs:
    row['valid_data'] = row['status'] == 200 and not row['html']
    if row['valid_data']:
        with ZipFile(ARCHIVES / (row['case']+'.zip')) as arc:
            payload = arc.read(next(name for name in arc.namelist() if name.endswith(row['suffix'])))
        row['archive_sha256'] = hashlib.sha256(payload).hexdigest()
        row['same_bytes_as_archive'] = row['sha256'] == row['archive_sha256']
        assert row['same_bytes_as_archive']
original_header = (LEGACY/'alpha-4.1-src/volbl/volbl.h').read_text()
modified_header = Path('/tmp/opencastp_float_metric/volbl.h').read_text()
assert original_header.replace('typedef double\tVol_real;', 'typedef float\tVol_real;') == modified_header
assert (LEGACY/'alpha-4.1-src/volbl/metric.c').read_bytes() == Path('/tmp/opencastp_float_metric/metric.c').read_bytes()
report = {
    'completed': True, 'passed': False, 'complete_server_equivalence': False,
    'assembled_utc': datetime.now(timezone.utc).isoformat(),
    'python': sys.version.split()[0],
    'source_commit': '4ffb6cbf3273a0cf22c42f700137ee96e2eead8a',
    'scope': 'bounded SA input/export diagnostics; no production numerical change',
    'region_comparator_changed': False, 'region_scalar_tolerance': 0.00050001,
    'modern_export_source_recovered': False,
    'producer_versions': {name: metadata.version(name) for name in ('opencastp', 'topomt', 'numpy', 'scipy', 'pyunitwizard')},
    'prepared_inputs_origin': 'trusted local frozen snapshots from the preceding Python 3.13 diagnostic; recalculations here use Python 3.14.7',
    'historical_pipeline_executed': False,
    'region_input_controls': controls,
    'atom_contributions': atoms,
    'atom588_metric_control': read('opencastp_atom588_metric_probe.json'),
    'atom588_input_control': read('opencastp_atom588_materialization.json'),
    'atom_materialization_panel': read('opencastp_sf_materialization_panel.json'),
    'live_output_inspection': outputs,
    'collector_sha256': {name+'.txt': hashlib.sha256(Path('/tmp', name).read_bytes()).hexdigest() for name in collectors},
    'historical_source_sha256': {name: hashlib.sha256((LEGACY/name).read_bytes()).hexdigest() for name in [
        'alpha-4.1-src/volbl/metric.c', 'alpha-4.1-src/volbl/volbl.h',
        'alpha-4.1-src/volbl/volbl.c', 'alpha-4.1-src/volbl/pdb2alf.c',
        'alpha-4.1-src/mkalf/alf.c', 'topomt_version/include/alf.h',
    ]},
    'modified_vol_real_header_sha256': hashlib.sha256(modified_header.encode()).hexdigest(),
}
assert report['atom_materialization_panel']['completed']
Path('/tmp/opencastp_input_export_controls_2026_10_03.json').write_text(json.dumps(report, indent=2)+'\n')
print('Assembled bounded controls; complete_server_equivalence=False', flush=True)

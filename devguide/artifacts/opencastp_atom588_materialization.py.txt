"""Ablate historical input materialization for one unresolved SA atom value."""

import ctypes as C
import hashlib
import json
import math
import pickle
import sys
from pathlib import Path
from importlib import metadata
from zipfile import ZipFile

import numpy as np

from opencastp._core import volbl
from opencastp._core.geometry import (
    ALF_VERTEX, ALF_EDGE, ALF_TRIANGLE, ALF_TETRA,
    _weighted_hidden0, _weighted_hidden1,
)

geometry = pickle.loads(Path('/tmp/opencastp_1mrg_geometry_314.pickle').read_bytes())
with ZipFile('/home/diego/repos@uibcdf/topomt/topomt/data/CASTpFold_server/1mrg.zip') as arc:
    raw_pdb = arc.read(next(n for n in arc.namelist() if n.endswith('.pdb')))
line = next(line for line in raw_pdb.decode().splitlines() if line.startswith('ATOM') and int(line[6:11]) == 588)
point = np.array([float(line[a:b]) for a, b in [(30, 38), (38, 46), (46, 54)]])
(target,) = np.flatnonzero(np.all(np.abs(geometry.atom_coordinates-point) < 1e-12, axis=1))
target = int(target)
edges = volbl._edge_atoms_by_master_id(geometry)
faces = volbl._face_atoms_by_master_id(geometry)
terms = []
for rank in range(1, geometry.base_rank+1):
    bounds = geometry.master_rank_offsets.get(rank)
    if bounds is None:
        continue
    for entry in geometry.master_entries[bounds[0]:bounds[1]]:
        kind = int(entry.f_type)
        if kind != ALF_TETRA and not entry.is_first:
            continue
        if kind == ALF_VERTEX:
            atoms = [int(entry.index)]
        elif kind == ALF_EDGE:
            atoms = list(edges[int(entry.index)])
        elif kind == ALF_TRIANGLE:
            atoms = list(faces[int(entry.index)])
        else:
            atoms = list(map(int, geometry.mesh.simplex_atom_indices[int(entry.index)]))
        if target in atoms:
            ordered = [target, *[a for a in atoms if a != target]]
            terms.append((len(atoms)-1, [*ordered, *[0]*(4-len(ordered))]))

H0 = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int)
H1 = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int, C.c_int)
CCW = C.CFUNCTYPE(C.c_int, C.c_int, C.c_int, C.c_int, C.c_int)
P = C.POINTER(C.c_double)
lib = C.CDLL('/tmp/opencastp_historical_sf.so')
lib.diagnostic_initialize.argtypes = [C.c_int, P, P, H0, H1, CCW]
lib.diagnostic_sf_term.argtypes = [C.c_int, C.c_int, C.c_int, C.c_int, C.c_int, P]
original = geometry.atom_coordinates
radii = geometry.atom_radii
weights = radii**2
h0 = H0(lambda i, j: int(_weighted_hidden0(original[i], float(weights[i]), original[j], float(weights[j])) != 0))
h1 = H1(lambda i, j, k: int(_weighted_hidden1(original[[i, j]], weights[[i, j]], original[k], float(weights[k])) != 0))
ccw = CCW(lambda i, j, k, l: int(np.linalg.det(np.array([original[j]-original[i], original[k]-original[i], original[l]-original[i]])) < 0))
scale = np.float32(1e-5)
fixed_points = np.rint(original*100000).astype(np.int64)
fixed_radii = np.rint(radii*100000).astype(np.int64)
legacy_points = (scale*fixed_points.astype(np.float32)).astype(np.float64)
legacy_radii = (scale*np.sqrt((fixed_radii**2).astype(np.float32))).astype(np.float64)
variants = {
    'baseline': (original, radii),
    'coordinates_direct_float32': (original.astype(np.float32).astype(float), radii),
    'radii_direct_float32': (original, radii.astype(np.float32).astype(float)),
    'both_direct_float32': (original.astype(np.float32).astype(float), radii.astype(np.float32).astype(float)),
    'coordinates_alf_get_coords_float32': (legacy_points, radii),
    'radii_alf_get_coords_float32': (original, legacy_radii),
    'both_alf_get_coords_float32': (legacy_points, legacy_radii),
}
rows = []
for variant, (points, sizes) in variants.items():
    points = np.ascontiguousarray(points, dtype=np.float64)
    sizes = np.ascontiguousarray(sizes, dtype=np.float64)
    lib.diagnostic_initialize(len(sizes), points.ctypes.data_as(P), sizes.ctypes.data_as(P), h0, h1, ccw)
    values = []
    for kind, indices in terms:
        out = np.zeros(2)
        lib.diagnostic_sf_term(kind, *indices, out.ctypes.data_as(P))
        values.append(float(out[0]))
    value = math.fsum(values)
    exported = float(np.round(float(f'{value:.4f}'), 3))
    rows.append({
        'variant': variant, 'volume_sa_angstrom3': value,
        'strict_passed': abs(value-25.886) <= 0.00050001,
        'diagnostic_export': exported, 'export_matches': exported == 25.886,
        'corrections': lib.diagnostic_corrections(), 'warnings': lib.diagnostic_warnings(),
    })
report = {
    'case': '1mrg', 'atom_id': 588, 'expected_volume_sa_angstrom3': 25.886,
    'scope': 'one atom; unchanged historical C SA primitives, fixed modern topology/predicates',
    'python': sys.version.split()[0],
    'versions': {name: metadata.version(name) for name in ('numpy', 'scipy', 'opencastp')},
    'source_commit': '4ffb6cbf3273a0cf22c42f700137ee96e2eead8a',
    'complete_server_equivalence': False, 'primitive_count': len(terms), 'variants': rows,
    'prepared_coordinate_sha256': hashlib.sha256(original.tobytes()).hexdigest(),
    'prepared_expanded_radius_sha256': hashlib.sha256(radii.tobytes()).hexdigest(),
}
Path('/tmp/opencastp_atom588_materialization.json').write_text(json.dumps(report, indent=2)+'\n')
print(json.dumps(report, indent=2), flush=True)

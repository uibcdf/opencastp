"""Check minimal SA volumes by independent planar sections before server results."""

import hashlib
import json
import pickle
import sys
from pathlib import Path

import numpy as np
import pyunitwizard as puw

sys.path.insert(0, '/home/diego/repos@uibcdf/opencastp')
from devtools.section_volume_oracle import integrate_tetrahedron

DIRECTORY = Path('/tmp/opencastp-minimal-server-controls')
state = json.loads((DIRECTORY / 'jobs.json').read_text())
report = dict(completed=False, complete_server_equivalence=False, geometry_and_sphere_inputs_shared=True,
              error_bound_certified=False, units={'length': 'angstrom', 'volume': 'angstrom**3'},
              python=sys.version.split()[0], collector_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
              section_oracle_sha256=hashlib.sha256(Path('/home/diego/repos@uibcdf/opencastp/devtools/section_volume_oracle.py').read_bytes()).hexdigest(), cases=[])
for job in state['jobs']:
    if any(feature['feature_type'] != 'void' for feature in job['native_predictions']):
        continue
    path = Path(job['native_cache_path'])
    assert hashlib.sha256(path.read_bytes()).hexdigest() == job['native_cache_sha256']
    result = pickle.loads(path.read_bytes())
    geometry = result.geometry
    coordinates = geometry.atom_coordinates
    radii = geometry.atom_radii
    assert len(result.features) == 1 and result.features[0]['feature_type'] == 'void'
    feature = result.features[0]
    variants = []
    for axis, divisions in ((2, 32), (0, 64)):
        order = [index for index in range(3) if index != axis] + [axis]
        integrals = []
        for index in feature['tetrahedron_indices']:
            atoms = geometry.mesh.simplex_atom_indices[index]
            tetra = coordinates[atoms]
            weights = radii[atoms] ** 2
            center = np.linalg.solve(2 * (tetra[1:] - tetra[0]),
                                     (np.sum(tetra[1:] ** 2, axis=1) - weights[1:]) - (np.sum(tetra[0] ** 2) - weights[0]))
            power = float(np.sum((center - tetra[0]) ** 2) - weights[0])
            assert power > 0
            bound_radius = np.sqrt(power) + 1e-5
            box = np.array([center - bound_radius, center + bound_radius])
            integrals.append(integrate_tetrahedron(tetra[:, order], coordinates[:, order], radii,
                                                 epsabs=1e-12, subdivisions=divisions, enclosure=box[:, order]))
        value = sum(entry['volume'] for entry in integrals)
        native = float(puw.get_value(feature['solvent_accessible_volume'], to_unit='angstrom**3'))
        variants.append(dict(axis=axis, subdivisions=divisions, volume_angstrom3=value,
                             native_volume_angstrom3=native, difference_from_native_angstrom3=value - native,
                             estimated_absolute_error_angstrom3=sum(entry['estimated_absolute_error'] for entry in integrals),
                             messages=[message for entry in integrals for message in entry['messages']], tetrahedra=len(integrals)))
        print(job['case'], axis, value, value - native, variants[-1]['estimated_absolute_error_angstrom3'], flush=True)
    report['cases'].append(dict(case=job['case'], pdb_sha256=job['pdb_sha256'], native_cache_sha256=job['native_cache_sha256'], variants=variants))
    (DIRECTORY / 'section_comparison_extended.json').write_text(json.dumps(report, indent=2) + '\n')
report['completed'] = True
(DIRECTORY / 'section_comparison_extended.json').write_text(json.dumps(report, indent=2) + '\n')

def load(path):
    return json.loads(path.read_text())


def append(path, heading, text):
    old = path.read_text()
    assert heading not in old, 'Preserve existing checkpoint on replay'
    path.write_text(old.rstrip() + '\n\n' + heading + '\n\n' + text.strip() + '\n')


state = load(DISCRIMINATOR / 'jobs.json')
comparison = load(DISCRIMINATOR / 'comparison.json')
assert state['completed'] and comparison['completed']
server = {row['case']: row['server_regions'][0] for row in comparison['cases']}
fields = dict(volume_sa='solvent_accessible_volume', volume_ms='molecular_surface_volume',
              area_sa='solvent_accessible_area', area_ms='molecular_surface_area')
verdicts = []
for name in ('baseline', 'disk_area_float32', 'cap_volume_decimal6'):
    checks = []
    for job in state['jobs']:
        predicted = job['predeclared_predictions'][name]
        for field, key in fields.items():
            actual = predicted[field]
            printed = server[job['case']]['metrics'][key]
            checks.append(dict(case=job['case'], field=field, actual=actual, printed=printed,
                               raw_pass=abs(actual - float(printed)) <= .00050001,
                               format_pass=f'{actual:.3f}' == printed))
    verdicts.append(dict(name=name, required=len(checks), raw_matches=sum(row['raw_pass'] for row in checks),
                         printed_matches=sum(row['format_pass'] for row in checks),
                         refuted_as_uniform_policy=not all(row['raw_pass'] for row in checks), checks=checks))
assert [row['raw_matches'] for row in verdicts] == [8, 7, 7]
